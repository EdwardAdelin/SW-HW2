### 1. List all books and their reading levels
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX book: <http://www.semanticweb.org/homework/bookrec#>
SELECT ?book ?level
WHERE {
  ?book rdf:type book:Book .
  ?book book:suitableFor ?level .
}

### 2. Find all themes associated with the book "Dune"
PREFIX book: <http://www.semanticweb.org/homework/bookrec#>
SELECT ?theme
WHERE {
  book:Dune book:hasTheme ?theme .
}

### 3. Find Alice's preferred theme and reading level
PREFIX book: <http://www.semanticweb.org/homework/bookrec#>
SELECT ?theme ?level
WHERE {
  book:Alice book:prefersTheme ?theme .
  book:Alice book:hasReadingLevel ?level .
}
